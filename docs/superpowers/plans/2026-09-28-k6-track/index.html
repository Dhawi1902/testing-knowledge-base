k6 Track Implementation Plan

For agentic workers: REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (- [ ]) syntax for tracking.

Goal: Add a k6 docs track (docs/k6/, 10 pages + index + 99) backed by a runnable sample (k6/sample/) that tests the Pizza Playground.

Architecture: Five k6 scripts share three small modules (lib/config.js, lib/auth.js, lib/flow.js) plus a CSV. Doc pages quote those files verbatim. The site gets a new sidebar and the k6 "coming soon" card becomes a live track, following commit b24053f.

Tech Stack: k6 v2.2.0 (plain JS modules, no npm), Pizza Playground (Docker Compose, API on :8080), Docusaurus 3.

Spec: docs/superpowers/specs/2026-09-28-k6-track-design.md

Global Constraints​

  • k6 version: v2.2.0, stated on the index page.
  • Plain JavaScript as k6 runs it. No bundler, no npm, no TypeScript.
  • BASE_URL defaults to http://127.0.0.1:8080.
  • Sample lives in k6/sample/. No docs copy in k6/docs/; docs/k6/ is the only copy of the pages.
  • performance-testing/03-mapping-to-k6.md is unchanged except one link line.
  • Every doc page: front matter (title, description, tags), Table of Contents with anchor links, Why → How → Tips, first person, practical tone.
  • Missing screenshots are marked <!-- TODO: Screenshot - description -->; images go in docs/k6/images/, never the repo root.
  • Every code block that shows a sample file is copied verbatim from k6/sample/.
  • Test accounts: owner@playground.local and customer01..20@playground.local, password Password123!.

Review Focus​

  1. Playground not running → every script stops in setup() with a message that says to run docker compose up, not a wall of status-0 errors. Pinned in Task 1 Step 5.
  2. More VUs than customers (21+) → VUs wrap around the 20 CSV rows instead of crashing on undefined. Pinned in Task 2 Step 4.
  3. Typo in PROFILE or EXECUTOR → k6 fails at init with the list of valid names instead of silently running 1 VU. Pinned in Task 4 Step 3.
  4. Gateway path (AUTOPAY=false) with MOCK_DELAY → the order still ends up PAID and order_to_paid shows the delay. Pinned in Task 3 Step 3.
  5. Reset really resets → after a run of N orders, the highest order number is 1000+N. Pinned in Task 4 Step 5.

// Settings shared by every script. Override any of them with -e NAME=value.
export const BASE_URL = __ENV.BASE_URL || 'http://127.0.0.1:8080';
export const API = `${BASE_URL}/api/v1`;

// Expected peak from the NFR. Small by default so a laptop can run it.
export const PEAK = Number(__ENV.USERS || 10);

// SCALE shrinks every duration for a dry run: SCALE=0.01 turns a 4h soak into about 2.5 minutes.
export const SCALE = Number(__ENV.SCALE || 1);
export const t = (seconds) => `${Math.max(1, Math.round(seconds * SCALE))}s`;

// The six test types from the Performance Testing module, as k6 stages.
// "peak" is VUs for ramping-vus, or iterations per second for ramping-arrival-rate.
export function stages(profile, peak) {
  const p = {
    smoke: [{ duration: t(60), target: Math.min(5, peak) }],
    load: [
      { duration: t(30), target: peak },
      { duration: t(1800), target: peak },
      { duration: t(60), target: 0 },
    ],
    stress: [
      { duration: t(120), target: Math.round(peak * 1.2) }, { duration: t(300), target: Math.round(peak * 1.2) },
      { duration: t(120), target: Math.round(peak * 1.5) }, { duration: t(300), target: Math.round(peak * 1.5) },
      { duration: t(120), target: peak * 2 }, { duration: t(300), target: peak * 2 },
      { duration: t(120), target: 0 },
    ],
    endurance: [
      { duration: t(600), target: peak },
      { duration: t(14400), target: peak },
      { duration: t(300), target: 0 },
    ],
    spike: [
      { duration: t(120), target: Math.round(peak / 2) },
      { duration: t(5), target: peak * 2 },
      { duration: t(180), target: peak * 2 },
      { duration: t(5), target: Math.round(peak / 2) },
      { duration: t(180), target: Math.round(peak / 2) },
    ],
  }[profile];
  if (!p) throw new Error(`Unknown PROFILE "${profile}". Use smoke, load, stress, endurance or spike.`);
  return p;
}

import http from 'k6/http';
import { API } from './config.js';

// Stop the test with a readable message instead of hundreds of failed requests.
export function expectOk(res, what) {
  if (res.status === 0) throw new Error(`${what}: no response from ${API}. Is the playground running? (docker compose up -d)`);
  if (res.status >= 400) throw new Error(`${what}: HTTP ${res.status} ${res.body}`);
  return res;
}

export function login(email, password) {
  const res = http.post(`${API}/auth/login`, JSON.stringify({ email, password }), {
    headers: { 'Content-Type': 'application/json' },
    tags: { name: 'POST /auth/login' },
  });
  return expectOk(res, `login ${email}`).json('token');
}

export function authHeaders(token) {
  return { 'Content-Type': 'application/json', Authorization: `Bearer ${token}` };
}

// Clean slate: clears orders and payments, order numbers restart at 1001.
export function resetPlayground() {
  const token = login('owner@playground.local', 'Password123!');
  expectOk(http.post(`${API}/admin/reset`, null, { headers: authHeaders(token) }), 'reset');
  return token;
}

import http from 'k6/http';
import { check, sleep } from 'k6';
import { API } from './config.js';
import { authHeaders } from './auth.js';

const pick = (list) => list[Math.floor(Math.random() * list.length)];

export function getMenu() {
  const res = http.get(`${API}/menu`, { tags: { name: 'GET /menu' } });
  check(res, { 'menu 200': (r) => r.status === 200 });
  return res.json();
}

// Correlation: ids come from the menu response, not hard-coded.
export function createOrder(token, customer, menu, { autoPay = true, mockDelay } = {}) {
  const body = {
    type: 'PICKUP',
    customer: { name: 'k6 load test', email: customer.email, phone: '0100000000' },
    items: [{
      pizzaId: pick(menu.pizzas).id,
      sizeId: pick(menu.sizes).id,
      crustId: pick(menu.crusts).id,
      toppingIds: [],
      quantity: 1,
    }],
    autoPay,
  };
  const headers = authHeaders(token);
  if (mockDelay) headers['X-Mock-Delay'] = String(mockDelay);
  const res = http.post(`${API}/orders`, JSON.stringify(body), { headers, tags: { name: 'POST /orders' } });
  const ok = check(res, {
    'order 201': (r) => r.status === 201,
    'order number assigned': (r) => r.status === 201 && r.json('orderNumber') >= 1001,
  });
  return ok ? res.json() : null; // { orderId, orderNumber, payUrl }
}

// What the browser does on the mock pay page. redirects: 0 stops k6 following
// the 302 to the web app, which a load test does not need.
export function payOnGateway(payUrl) {
  const res = http.post(payUrl, { action: 'pay' }, { redirects: 0, tags: { name: 'POST /pay/{billId}' } });
  check(res, { 'gateway accepted payment': (r) => r.status === 302 });
}

export function waitUntilPaid(orderId, tries = 10) {
  for (let i = 0; i < tries; i++) {
    const res = http.get(`${API}/payments/${orderId}/status`, { tags: { name: 'GET /payments/{id}/status' } });
    if (res.status === 200 && res.json('paymentStatus') === 'PAID') return true;
    sleep(0.5);
  }
  return false;
}

Note the tags: { name: ... } with {billId} / {id}: without it every order id becomes its own URL in the results.

import { group, sleep } from 'k6';
import { Trend, Counter } from 'k6/metrics';
import { login, resetPlayground } from './lib/auth.js';
import { customerFor } from './lib/data.js';
import { getMenu, createOrder, payOnGateway, waitUntilPaid } from './lib/flow.js';

const orderToPaid = new Trend('order_to_paid', true); // true = time values
const ordersPlaced = new Counter('orders_placed');
const AUTOPAY = __ENV.AUTOPAY !== 'false';

export const options = {
  vus: 3,
  duration: '30s',
  thresholds: {
    checks: ['rate==1'],
    http_req_failed: ['rate<0.005'],                           // NFR: error rate <= 0.5%
    'http_req_duration{name:GET /menu}': ['p(95)<300'],        // per endpoint, via the name tag
    'http_req_duration{name:POST /orders}': ['p(95)<1000'],
    'group_duration{group:::order}': ['avg<3000'],             // NFR: whole step <= 3s
    order_to_paid: ['p(95)<5000'],
    // Stop early instead of burning an hour on a run that already failed.
    http_req_duration: [{ threshold: 'p(99)<3000', abortOnFail: true, delayAbortEval: '10s' }],
  },
};

export function setup() {
  resetPlayground();
}

let token;
export default function () {
  const me = customerFor(__VU);
  if (!token) token = login(me.email, me.password);

  let menu;
  group('browse', () => {
    menu = getMenu();
  });
  sleep(1); // think time: reading the menu

  group('order', () => {
    const start = Date.now();
    const order = createOrder(token, me, menu, { autoPay: AUTOPAY });
    if (!order) return;
    if (!AUTOPAY) {
      payOnGateway(order.payUrl);
      waitUntilPaid(order.orderId);
    }
    orderToPaid.add(Date.now() - start);
    ordersPlaced.add(1);
  });
  sleep(1);
}

import http from 'k6/http';
import { check, sleep } from 'k6';
import { API, PEAK, t, stages } from './lib/config.js';
import { login, authHeaders, resetPlayground } from './lib/auth.js';
import { customerFor } from './lib/data.js';
import { getMenu, createOrder } from './lib/flow.js';

const PROFILE = __ENV.PROFILE || 'smoke';
const EXECUTOR = __ENV.EXECUTOR || 'ramping-vus';
const RATE = Number(__ENV.RATE || 5);     // orders per second, for arrival-rate executors
const THINK = Number(__ENV.THINK ?? 1);   // seconds; set 0 with arrival-rate executors

const scenarios = {
  // Fixed amount of work, split across VUs (fast VUs do more).
  'shared-iterations': { executor: 'shared-iterations', vus: PEAK, iterations: PEAK * 10, maxDuration: t(600) },
  // Every VU does the same number of iterations.
  'per-vu-iterations': { executor: 'per-vu-iterations', vus: PEAK, iterations: 10, maxDuration: t(600) },
  // N users for a fixed time. Closest to a plain JMeter Thread Group.
  'constant-vus': { executor: 'constant-vus', vus: PEAK, duration: t(300) },
  // Users follow the PROFILE shape. This is how smoke/load/stress/endurance/spike run.
  'ramping-vus': { executor: 'ramping-vus', startVUs: 0, stages: stages(PROFILE, PEAK), gracefulRampDown: '30s' },
  // A fixed request rate, whatever the response time. For "200 TPS" style NFRs.
  'constant-arrival-rate': {
    executor: 'constant-arrival-rate', rate: RATE, timeUnit: '1s', duration: t(300),
    preAllocatedVUs: PEAK, maxVUs: PEAK * 5,
  },
  // The PROFILE shape, but in orders per second instead of users.
  'ramping-arrival-rate': {
    executor: 'ramping-arrival-rate', startRate: 0, timeUnit: '1s', stages: stages(PROFILE, RATE),
    preAllocatedVUs: PEAK, maxVUs: PEAK * 5,
  },
};

// Two populations in one run: browsers who never order, and buyers at a fixed rate.
const mixed = {
  browsers: { executor: 'ramping-vus', stages: stages(PROFILE, PEAK), exec: 'browse' },
  buyers: {
    executor: 'constant-arrival-rate', rate: RATE, timeUnit: '1s', duration: t(300),
    preAllocatedVUs: PEAK, maxVUs: PEAK * 5, exec: 'buy',
  },
};

function pickScenarios() {
  if (EXECUTOR === 'mixed') return mixed;
  if (!scenarios[EXECUTOR]) {
    throw new Error(`Unknown EXECUTOR "${EXECUTOR}". Use ${Object.keys(scenarios).join(', ')} or mixed.`);
  }
  return { [EXECUTOR]: { ...scenarios[EXECUTOR], exec: 'buy' } };
}

export const options = {
  scenarios: pickScenarios(),
  thresholds: {
    http_req_duration: ['avg<3000'],  // NFR: average response time <= 3s
    http_req_failed: ['rate<0.005'],  // NFR: error rate <= 0.5%
  },
};

export function setup() {
  resetPlayground();
}

export function browse() {
  getMenu();
  sleep(THINK);
}

let token;
export function buy() {
  const me = customerFor(__VU);
  if (!token) token = login(me.email, me.password);
  createOrder(token, me, getMenu(), { autoPay: true });
  sleep(THINK);
}
